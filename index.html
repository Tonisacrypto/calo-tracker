function doGet(e) {
  var props = PropertiesService.getScriptProperties();
  var savedState = props.getProperty("NUTRICHAT_CLOUD_STATE") || "{}";
  return ContentService.createTextOutput(savedState).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    var body = JSON.parse(e.postData.contents);
    var props = PropertiesService.getScriptProperties();
    if (body.fullState) {
      props.setProperty("NUTRICHAT_CLOUD_STATE", JSON.stringify(body.fullState));
    }
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    if (body.daily) {
      var d = body.daily;
      var sh = ss.getSheetByName("Nhat_Ky_Hang_Ngay") || ss.getSheets()[1];
      var data = sh.getDataRange().getValues();
      var foundRow = -1;
      for (var i = 5; i < data.length; i++) {
        var cellDate = data[i][0];
        var strDate = (cellDate instanceof Date) ? Utilities.formatDate(cellDate, Session.getScriptTimeZone(), "yyyy-MM-dd") : String(cellDate).trim();
        if (strDate === d.date || strDate === "") {
          foundRow = i + 1;
          break;
        }
      }
      if (foundRow === -1) foundRow = sh.getLastRow() + 1;
      sh.getRange(foundRow, 1, 1, 4).setValues([[d.date, d.modeLabel, d.steps, d.gymMins]]);
      sh.getRange(foundRow, 10, 1, 1).setValue(d.intake);
      sh.getRange(foundRow, 12, 1, 1).setValue(d.p + "g P | " + d.c + "g C | " + d.f + "g F | " + d.water + "ml");
    }
    return ContentService.createTextOutput(JSON.stringify({status: "ok"})).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}
