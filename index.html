Bước 1: Kết nối Google Apps Script để Đồng bộ Tự động giữa Samsung & Máy tính

Vì bạn nhập liệu trên cả điện thoại Samsung và máy tính, việc gắn đường dẫn Web App URL dưới đây sẽ biến Google Sheets thành bộ nhớ đám mây chung: nhập ở điện thoại xong mở máy tính sẽ tự động cập nhật (và ngược lại), đồng thời tự ghi vào bảng Nhat_Ky_Hang_Ngay và Do_Eo_Hong_Hang_Tuan. 

// MÃ GOOGLE APPS SCRIPT ĐỒNG BỘ 2 CHIỀU (SAMSUNG <-> MÁY TÍNH <-> GOOGLE SHEETS)
function doGet(e) {
  var props = PropertiesService.getScriptProperties();
  var savedState = props.getProperty("NUTRICHAT_CLOUD_STATE") || "{}";
  return ContentService.createTextOutput(savedState).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    var body = JSON.parse(e.postData.contents);
    var props = PropertiesService.getScriptProperties();
    if (body.fullState) {
      props.setProperty("NUTRICHAT_CLOUD_STATE", JSON.stringify(body.fullState));
    }
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    if (body.daily) {
      var d = body.daily;
      var sh = ss.getSheetByName("Nhat_Ky_Hang_Ngay") || ss.getSheets()[1];
      var data = sh.getDataRange().getValues();
      var foundRow = -1;
      for (var i = 5; i < data.length; i++) {
        var cellDate = data[i][0];
        var strDate = (cellDate instanceof Date) ? Utilities.formatDate(cellDate, Session.getScriptTimeZone(), "yyyy-MM-dd") : String(cellDate).trim();
        if (strDate === d.date || strDate === "") {
          foundRow = i + 1;
          break;
        }
      }
      if (foundRow === -1) foundRow = sh.getLastRow() + 1;
      sh.getRange(foundRow, 1, 1, 4).setValues([[d.date, d.modeLabel, d.steps, d.gymMins]]);
      sh.getRange(foundRow, 10, 1, 1).setValue(d.intake);
      sh.getRange(foundRow, 12, 1, 1).setValue(d.p + "g P | " + d.c + "g C | " + d.f + "g F | " + d.water + "ml");
    }
    if (body.weeklyBody) {
      var shW = ss.getSheetByName("Do_Eo_Hong_Hang_Tuan") || ss.getSheets()[2];
      for (var w = 1; w <= 12; w++) {
        var wk = body.weeklyBody[String(w)];
        if (wk) {
          var r = w + 5;
          if (wk.date) shW.getRange(r, 2).setValue(wk.date);
          if (wk.weight !== "") shW.getRange(r, 3).setValue(wk.weight);
          if (wk.waist !== "") shW.getRange(r, 5).setValue(wk.waist);
          if (wk.hip !== "") shW.getRange(r, 7).setValue(wk.hip);
        }
      }
    }
    return ContentService.createTextOutput(JSON.stringify({status: "ok", syncedAt: new Date().toISOString()})).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}
